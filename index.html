قالب SQL جاهز لـ Supabase (RLS + منع التضارب)

-- Enable RLS on core tables
alter table if exists public.salon_data enable row level security;
alter table if exists public.salon_bookings enable row level security;
alter table if exists public.salon_customers enable row level security;

-- Public read policy for salon catalog/settings
drop policy if exists "public_read_salon_data" on public.salon_data;
create policy "public_read_salon_data" on public.salon_data for select using (true);

-- Allow booking inserts/updates via client with validation
drop policy if exists "public_insert_salon_bookings" on public.salon_bookings;
create policy "public_insert_salon_bookings" on public.salon_bookings for insert with check (true);
drop policy if exists "public_select_salon_bookings" on public.salon_bookings;
create policy "public_select_salon_bookings" on public.salon_bookings for select using (true);
drop policy if exists "public_update_salon_bookings" on public.salon_bookings;
create policy "public_update_salon_bookings" on public.salon_bookings for update using (true);
drop policy if exists "public_delete_salon_bookings" on public.salon_bookings;
create policy "public_delete_salon_bookings" on public.salon_bookings for delete using (true);
